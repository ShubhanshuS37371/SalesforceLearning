    Image = TILES + '/affordable.jpeg';
    URL;
    connectedCallback() {
        this.URL = window.location.origin;
        console.log('Base URL: ' + this.URL);
    }

    @wire(getTilesData)
    wiredTiles({ error, data }) {
        if (data) {
            console.log('Fetched Data: --->', data);
            this.buildings = data;
            this.buildingimage = this.buildings[0].imageUrl;
            console.log('builidings image===='+this.buildingimage);
            this.TilesLink = TILES+'/'+this.buildingimage;
            console.log("image==", this.TilesLink);
            
            // this.buildings = data.map(tile => ({
                // ...tile,
                // imageUrl: tile.SH_Image_Url__c ? `${TILES}/${tile.SH_Image_Url__c}` : '',
                //imageUrl: tile.imageUrl ? `${TILES}/${tile.imageUrl}` : `${TILES}/affordable.jpeg`,
                // communitypropUrl: `/details-page/?recordId=${tile.Id}`
            // }));
            this.buildings = data.map(tile => ({
                ...tile,
                image: tile.imageUrl ? `${TILES}/${tile.imageUrl}` : ''
            }));
            
            console.log('Fetched buildings: ', this.buildings);
        } else if (error) {
            console.error('Error fetching event buildings:', error);
        }
        console.log('Processed Tiles buildings: ', JSON.stringify(this.buildings));
    }

    @wire(getEventTilesprop)
    wiredEventTilesprop({ error, data }) {
        if (data) {
            console.log('Fetched Data property: ', data);
            this.tileprops = data.map(tileprop => ({
                id: tileprop.Id,
                name: tileprop.Name,
                noofunits: tileprop.SH_No_of_units__c,
                noofunitsvacant: tileprop.SH_No_of_Units_Vacant__c,
                noofroom: tileprop.SH_Number_of_Room__c,
                communitypropUrl: '/details-page/?recordId='+tileprop.Id
            }));
            console.log('Processed Tileprops: ', JSON.stringify(this.tilesprops));
        } else if (error) {
            console.error('Error fetching event buildings:', error);
        }
    }

    handleTileClick(event) {
        const tileId = event.currentTarget.dataset.id; 
        const selectedTile = this.buildings.find(tile => tile.id === tileId);
        console.log('tileId===='+selectedTile);
        if (!selectedTile) {
            console.error('Tile not found.');
            return;
        }

        const { label: eventtype, communityURL } = selectedTile;

        if (!communityURL) {
            console.error('Community URL is missing for this tile.');
            return;
        }

        this[NavigationMixin.Navigate]({
            type: 'standard__webPage',
            attributes: {
                url: `${communityURL}?eventType=${eventtype}`
            }
        });

        this.dispatchEvent(new CustomEvent('tileclick', {
            detail: { eventtype }
        }));
    }


    handleClick(event) {
        console.log('This is the data set ::: ' + JSON.stringify(event.target.dataset));
        const propertyId = event.currentTarget.dataset.propertyId; 
        const buildingId = event.currentTarget.dataset.buildingId; 

        // Navigate to community page with URL params
        this[NavigationMixin.Navigate]({
            type: 'standard__webPage',
            attributes: {
                url: `/sh-property/${propertyId}?buildingid=${buildingId}`
            }
        });
    }
}

public class SH_TILECONFIG_ServicesController {
    @AuraEnabled(cacheable = true)
    public static List < SH_Building__c > getEventTiles() {
        try {
            List < SH_Building__c > eventList = new List < SH_Building__c > ();
            eventList = [SELECT Id, SH_Tile_Name__c, SH_Service_Status__c, SH_Address__c, SH_Image_Url__c, SH_Community_Page_Url__c, SH_Building_Name__c,SH_Tile_Subheading__c, SH_Order__c FROM SH_Building__c
                WHERE SH_Service_Status__c = 'Active'
                order by SH_Order__c asc
            ];
            return eventList;
        } catch (Exception e) {
            throw new AuraHandledException(e.getMessage() + '===>' + e.getLineNumber());
        }
    }

    //Get related properties of building
    @AuraEnabled(cacheable = true)
    public static List < SH_Property__c > getPropertyTilesForActiveBuildings() {
        try {
            // First, fetch active buildings
            List < SH_Building__c > activeBuildings = [
                SELECT Id
                FROM SH_Building__c
                WHERE SH_Service_Status__c = 'Active'
            ];

            Set < Id > buildingIds = new Set < Id > ();
            for (SH_Building__c b: activeBuildings) {
                buildingIds.add(b.Id);
            }

            List < SH_Property__c > propertyList = [
                SELECT Id, Name, SH_Building__r.Id, SH_Building__r.Name, SH_No_of_units__c, SH_No_of_Units_Vacant__c, SH_Number_of_Room__c
                FROM SH_Property__c
                WHERE SH_Building__c IN: buildingIds LIMIT 2
            ];

            return propertyList;

        } catch (Exception e) {
            throw new AuraHandledException(e.getMessage() + ' ===> Line: ' + e.getLineNumber());
        }
    }

    @AuraEnabled(cacheable = true)
    public static List < BuildingWrapper > getTilesData() {
        try {
            List < SH_Property__c > propertyList = [
                SELECT
                Id,
                Name,
                SH_Building__r.Id,
                SH_Building__r.Name,
                SH_Building__r.SH_Tile_Name__c,
                SH_Building__r.SH_Service_Status__c,
                SH_Building__r.SH_Address__c,
                SH_Building__r.SH_Image_Url__c,
                SH_Building__r.SH_Community_Page_Url__c,
                SH_Building__r.SH_Tile_Subheading__c,
                SH_Building__r.SH_Order__c,
                SH_Building__r.SH_Building_Name__c,
                SH_No_of_units__c,
                SH_No_of_Units_Vacant__c,
                SH_Number_of_Room__c,
                SH_Barrier_Free__c
                FROM SH_Property__c
                WHERE SH_Building__r.SH_Service_Status__c = 'Active'
                ORDER BY SH_Building__r.SH_Order__c ASC
            ];

            Map < Id, BuildingWrapper > buildingMap = new Map < Id, BuildingWrapper > ();

            for (SH_Property__c prop: propertyList) {
                Id bId = prop.SH_Building__r.Id;

                if (!buildingMap.containsKey(bId)) {
                    BuildingWrapper bw = new BuildingWrapper();
                    bw.buildingId = bId;
                    bw.buildingName = prop.SH_Building__r.Name;
                    bw.tileName = prop.SH_Building__r.SH_Tile_Name__c;
                    bw.serviceStatus = prop.SH_Building__r.SH_Service_Status__c;
                    bw.address = prop.SH_Building__r.SH_Address__c;
                    bw.imageUrl = prop.SH_Building__r.SH_Image_Url__c;
                    bw.building_Name = prop.SH_Building__r.SH_Building_Name__c;
                    bw.communityPageUrl = prop.SH_Building__r.SH_Community_Page_Url__c;
                    bw.tileSubheading = prop.SH_Building__r.SH_Tile_Subheading__c;
                    bw.order = String.valueOf(prop.SH_Building__r.SH_Order__c != null ? prop.SH_Building__r.SH_Order__c.intValue() : 0);
                    buildingMap.put(bId, bw);
                }
                BuildingWrapper bWrapper = buildingMap.get(bId);
                //LIMITTER
                if (bWrapper.properties.size() < 8) {
                    buildingMap.get(bId).properties.add(new PropertyWrapper(prop));
                }
            }

            return new List < BuildingWrapper > (buildingMap.values());

        } catch (Exception e) {
            throw new AuraHandledException(e.getMessage() + ' ===> Line: ' + e.getLineNumber());
        }
    }

    public class BuildingWrapper {
        @AuraEnabled public Id buildingId;
        @AuraEnabled public String buildingName;
        @AuraEnabled public String building_Name;
        @AuraEnabled public String tileName;
        @AuraEnabled public String serviceStatus;
        @AuraEnabled public String address;
        @AuraEnabled public String imageUrl;
        @AuraEnabled public String communityPageUrl;
        @AuraEnabled public String tileSubheading;
        @AuraEnabled public String order;
        @AuraEnabled public List < PropertyWrapper > properties = new List < PropertyWrapper > ();
    }
    public class PropertyWrapper {
        @AuraEnabled public Id propertyId;
        @AuraEnabled public String propertyName;
        @AuraEnabled public String noOfUnits;
        @AuraEnabled public String noOfUnitsVacant;
        @AuraEnabled public String numberOfRooms;
        @AuraEnabled public Boolean barrierFree;

        public PropertyWrapper(SH_Property__c prop) {
            this.propertyId = prop.Id;
            this.propertyName = prop.Name;
            this.barrierFree = prop.SH_Barrier_Free__c;
            this.noOfUnits = String.valueOf(prop.SH_No_of_units__c);
            this.noOfUnitsVacant = String.valueOf(prop.SH_No_of_Units_Vacant__c);
            this.numberOfRooms = String.valueOf(prop.SH_Number_of_Room__c);
        }
    }
}
